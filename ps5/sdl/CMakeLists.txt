cmake_minimum_required(VERSION 3.20)
project(mkw_ps5_sdl_input LANGUAGES C)
get_filename_component(MKW_WORKSPACE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
# Compile WiiCompiled's pinned SDL for its real gamepad/event/IO machinery.
# Native Pad samples will enter its virtual joystick driver. No SDL GPU path.
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
set(SDL_UNIX_CONSOLE_BUILD ON CACHE BOOL "" FORCE)
foreach(feature VIDEO GPU RENDER CAMERA AUDIO HAPTIC POWER SENSOR DIALOG TRAY
        DBUS IBUS LIBURING LIBUDEV HIDAPI_LIBUSB HIDAPI_JOYSTICK SYSTEM_ICONV LIBICONV)
    set(SDL_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
option(MKW_PS5_NATIVE_AUDIO "Compile the native AudioOut playback driver" OFF)
if(MKW_PS5_NATIVE_AUDIO)
    set(SDL_AUDIO ON CACHE BOOL "" FORCE)
    foreach(driver ALSA OSS SNDIO JACK PIPEWIRE PULSEAUDIO NETBSD WASAPI DSOUND DUMMYAUDIO DISKAUDIO)
        set(SDL_${driver} OFF CACHE BOOL "" FORCE)
    endforeach()
    set(SDL_AUDIO_DRIVER_PRIVATE 1)
    set(HAVE_SDL_AUDIO TRUE)
endif()
set(SDL_JOYSTICK ON CACHE BOOL "" FORCE)
set(SDL_HAPTIC ON CACHE BOOL "" FORCE)
# The upstream virtual-driver option depends on HIDAPI, although virtual input
# does not open USB hardware. All actual HID and USB joystick drivers are off.
set(SDL_HIDAPI ON CACHE BOOL "" FORCE)
set(SDL_VIRTUAL_JOYSTICK ON CACHE BOOL "" FORCE)
set(SDL_PTHREADS ON CACHE BOOL "" FORCE)
set(SDL_PTHREADS_SEM OFF CACHE BOOL "" FORCE)
set(LIBUSBHID OFF CACHE INTERNAL "PS5 input uses libScePad" FORCE)
set(HAVE_LINUX_INPUT_H OFF CACHE INTERNAL "PS5 input uses libScePad" FORCE)
set(SDL_PRESEED OFF CACHE BOOL "" FORCE)
if(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
    # FreeBSD headers describe the cross ABI, not the console's OS services.
    # Use SDL's private-platform hooks so shutdown does not pull in Unix GTK.
    set(SDL_PLATFORM_PRIVATE ON)
    # Header-only cross checks cannot prove PS5 has a POSIX dynamic loader.
    # Use SDL's static API and unsupported-loadso implementation until one exists.
    set(HAVE_DLOPEN_IN_LIBC OFF CACHE INTERNAL "No PS5 POSIX dynamic loader" FORCE)
    set(HAVE_DLOPEN_IN_LIBDL OFF CACHE INTERNAL "No PS5 POSIX dynamic loader" FORCE)
    set(HAVE_DLOPEN OFF CACHE INTERNAL "No PS5 POSIX dynamic loader" FORCE)
    # Native environment mutation and these wide-string helpers are unavailable.
    # Keep reads and writes in SDL's own environment, even though native getenv
    # exists. SDL derives HAVE_* from LIBC_HAS_* for these symbol checks.
    foreach(function GETENV SETENV UNSETENV WCSLCPY WCSLCAT WCSNLEN)
        set(LIBC_HAS_${function} OFF CACHE INTERNAL "Use SDL portable PS5 implementation" FORCE)
    endforeach()
    set(HAVE_GETHOSTNAME OFF CACHE INTERNAL "No PS5 gethostname export" FORCE)
endif()
add_subdirectory("${MKW_WORKSPACE}/.tools/game-deps/SDL3-3.4.4" sdl)
if(MKW_PS5_NATIVE_AUDIO)
    target_sources(SDL3-static PRIVATE "${CMAKE_CURRENT_LIST_DIR}/audio_ps5.c")
    target_include_directories(SDL3-static PRIVATE "${MKW_WORKSPACE}/.tools/game-deps/SDL3-3.4.4/src")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
    target_include_directories(SDL3-static PRIVATE "${CMAKE_CURRENT_LIST_DIR}/include")
    # Absolute file operations use the native POSIX exports tested on console.
    # Platform path discovery stays unsupported; the runtime supplies its roots.
    target_sources(SDL3-static PRIVATE "${CMAKE_CURRENT_LIST_DIR}/filesystem_ps5.c")
endif()

option(MKW_BUILD_VIRTUAL_PAD_TEST "Run the native snapshot bridge against host SDL" OFF)
option(MKW_BUILD_AUDIO_TEST "Test real runtime audio through a simulated native device" OFF)
if(MKW_BUILD_AUDIO_TEST)
    if(NOT MKW_PS5_NATIVE_AUDIO)
        message(FATAL_ERROR "Audio contract tests require the PS5 audio driver")
    endif()
    enable_language(CXX)
    add_executable(audio_backend_test ../tests/audio_backend.cpp "${MKW_WORKSPACE}/Wiicompiled/runtime/src/audio_backend.cpp")
    target_include_directories(audio_backend_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${MKW_WORKSPACE}/Wiicompiled/runtime/include")
    target_compile_features(audio_backend_test PRIVATE cxx_std_17)
    target_compile_definitions(audio_backend_test PRIVATE SDL_MAIN_HANDLED)
    target_compile_options(audio_backend_test PRIVATE -Wall -Wextra)
    target_link_libraries(audio_backend_test PRIVATE SDL3::SDL3-static)
endif()
if(MKW_BUILD_VIRTUAL_PAD_TEST)
    enable_language(CXX)
    add_executable(virtual_pad_test ../tests/virtual_pad.cpp ../input/virtual_pad.cpp)
    target_include_directories(virtual_pad_test PRIVATE ../input)
    target_compile_features(virtual_pad_test PRIVATE cxx_std_20)
    target_compile_options(virtual_pad_test PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(virtual_pad_test PRIVATE SDL3::SDL3-static)
    set(ABSL_BUILD_TESTING OFF CACHE BOOL "" FORCE)
    set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
    set(ABSL_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
    set(CMAKE_CXX_STANDARD 20)
    add_subdirectory("${MKW_WORKSPACE}/.tools/desktop-check/abseil-cpp-20240722.0" abseil)
    add_subdirectory("${MKW_WORKSPACE}/.tools/game-deps/fmt-11.1.4" fmt)
    target_compile_options(fmt PRIVATE -include cstdlib)
    set(aurora "${MKW_WORKSPACE}/Wiicompiled/aurora-main")
    add_executable(aurora_pad_test ../tests/aurora_pad.cpp ../input/virtual_pad.cpp
        "${aurora}/lib/input.cpp" "${aurora}/lib/dolphin/pad/pad.cpp" "${aurora}/lib/logging.cpp")
    target_include_directories(aurora_pad_test PRIVATE ../input "${aurora}/include" "${aurora}/lib")
    target_compile_definitions(aurora_pad_test PRIVATE TARGET_PC=1)
    target_compile_options(aurora_pad_test PRIVATE -include cstdlib)
    target_compile_features(aurora_pad_test PRIVATE cxx_std_20)
    target_link_libraries(aurora_pad_test PRIVATE SDL3::SDL3-static absl::flat_hash_map fmt::fmt)
    add_executable(aurora_input_test ../tests/aurora_input.cpp ../input/aurora_input.cpp ../input/pad_autopilot.cpp
        ../input/native_pad.cpp ../input/pad_sample.cpp ../input/virtual_pad.cpp
        "${aurora}/lib/input.cpp" "${aurora}/lib/dolphin/pad/pad.cpp" "${aurora}/lib/logging.cpp")
    target_include_directories(aurora_input_test PRIVATE ../input "${aurora}/include" "${aurora}/lib")
    target_compile_definitions(aurora_input_test PRIVATE TARGET_PC=1)
    target_compile_options(aurora_input_test PRIVATE -include cstdlib -Wall -Wextra)
    target_compile_features(aurora_input_test PRIVATE cxx_std_20)
    target_link_libraries(aurora_input_test PRIVATE SDL3::SDL3-static absl::flat_hash_map fmt::fmt)
endif()
